// Racing Logic: Extended Duration & Static Variance function startRace() { if (bornRacers.length === 0) return alert('BREED RACERS FIRST!'); const raceTrack = document.getElementById('race-track'); const raceResults = document.getElementById('race-results'); raceTrack.innerHTML = ''; raceResults.classList.remove('show'); document.getElementById('start-race-btn').disabled = true; const racers = [...bornRacers].slice(-8); // Increased the denominator and added a multiplier to stretch the race 3x const raceState = racers.map(r => ({ ...r, progress: 0, // Base speed derived from stats + a smaller random jitter for "form" step: (r.speed / 800) + (Math.random() * 0.05) })); raceState.forEach(r => { const lane = document.createElement('div'); lane.className = 'race-lane'; lane.innerHTML = `
${r.name}
0%
`; raceTrack.appendChild(lane); }); const timer = setInterval(() => { let done = 0; raceState.forEach(r => { // Movement logic updated for 3x duration r.progress += r.step; if (r.progress >= 100) { r.progress = 100; done++; } const bar = document.getElementById(`bar-${r.rank}`); const pos = document.getElementById(`pos-${r.rank}`); if (bar) bar.style.maxWidth = r.progress + '%'; if (pos) pos.innerText = Math.floor(r.progress) + '%'; }); if (done === raceState.length) { clearInterval(timer); document.getElementById('start-race-btn').disabled = false; // Sort by actual speed to verify the winner is the fastest const winners = raceState.sort((a, b) => b.speed - a.speed); raceResults.innerHTML = `
🏆 DOMINANT WINNER: ${winners[0].name}
TOP SPEED: ${winners[0].speed}
`; raceResults.classList.add('show'); } }, 60); // Slower tick rate for smoother visual tracking }
